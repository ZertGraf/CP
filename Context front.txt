import { useState } from 'react'
import { api } from '../api'

export default function Login({ onLogin }) {
    const [isRegister, setIsRegister] = useState(false)
    const [email, setEmail] = useState('')
    const [password, setPassword] = useState('')
    const [name, setName] = useState('')
    const [role, setRole] = useState('operator')
    const [error, setError] = useState('')

    async function handleSubmit(e) {
        e.preventDefault()
        setError('')
        try {
            if (isRegister) {
                await api.register(email, password, name, role)
            }
            const data = await api.login(email, password)
            localStorage.setItem('token', data.token)
            localStorage.setItem('role', data.role)
            localStorage.setItem('name', data.name)
            onLogin()
        } catch (err) {
            setError(err.message)
        }
    }

    return (
        <div style={{
            display: 'flex',
            flexDirection: 'column',
            alignItems: 'center',
            justifyContent: 'center',
            minHeight: '100vh',
            padding: 20
        }}>
            <h1 style={{ marginBottom: 32, fontSize: 36 }}>🌱 plantation control</h1>
            <div className="card" style={{ width: '100%', maxWidth: 420, padding: 32 }}>
                <h2 style={{ marginBottom: 20 }}>{isRegister ? 'регистрация' : 'вход'}</h2>
                {error && <p className="error">{error}</p>}
                <form onSubmit={handleSubmit} style={{ display: 'flex', flexDirection: 'column', gap: 12 }}>
                    {isRegister && (
                        <>
                            <input placeholder="имя" value={name}
                                   onChange={e => setName(e.target.value)}
                                   style={{ width: '100%' }} />
                            <select value={role} onChange={e => setRole(e.target.value)}
                                    style={{ width: '100%' }}>
                                <option value="operator">оператор</option>
                                <option value="agronomist">агроном</option>
                            </select>
                        </>
                    )}
                    <input type="email" placeholder="email" value={email}
                           onChange={e => setEmail(e.target.value)}
                           style={{ width: '100%' }} />
                    <input type="password" placeholder="пароль" value={password}
                           onChange={e => setPassword(e.target.value)}
                           style={{ width: '100%' }} />
                    <button className="primary" type="submit" style={{ width: '100%', marginTop: 8 }}>
                        {isRegister ? 'зарегистрироваться' : 'войти'}
                    </button>
                </form>
                <p style={{ marginTop: 16, fontSize: 14, textAlign: 'center' }}>
                    <span style={{ cursor: 'pointer', color: '#4a7c59' }}
                          onClick={() => setIsRegister(!isRegister)}>
                        {isRegister ? 'уже есть аккаунт? войти' : 'нет аккаунта? регистрация'}
                    </span>
                </p>
            </div>
        </div>
    )
}

import { useState, useEffect } from 'react'
import { api } from '../api'

export default function Plants() {
    const [plants, setPlants] = useState([])
    const [sectors, setSectors] = useState([])
    const [sectorId, setSectorId] = useState('')
    const [species, setSpecies] = useState('Личи')
    const [age, setAge] = useState('')
    const [filter, setFilter] = useState('')
    const [error, setError] = useState('')

    async function load() {
        try {
            const p = await api.getPlants(filter)
            setPlants(p || [])
            const s = await api.getSectors()
            setSectors(s || [])
        } catch (err) {
            console.error(err)
        }
    }

    useEffect(() => { load() }, [filter])

    async function handleCreate(e) {
        e.preventDefault()
        setError('')
        if (!sectorId) return setError('Выберите сектор для посадки!')

        try {
            await api.createPlant({ sector_id: sectorId, species: species || 'Личи', age_months: parseInt(age) || 0 })
            setSpecies('Личи')
            setAge('')
            load()
        } catch (err) {
            setError(err.message)
        }
    }

    async function handleDelete(id) {
        if (!confirm('Удалить это растение?')) return
        await api.deletePlant(id)
        load()
    }

    return (
        <div>
            <div style={{display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginBottom: 24}}>
                <h1 style={{marginBottom: 0}}>🌱 Растения</h1>
                <select value={filter} onChange={e => setFilter(e.target.value)} style={{maxWidth: 250, boxShadow: 'none'}}>
                    <option value="">Показать все секторы</option>
                    {sectors.map(s => (<option key={s.id} value={s.id}>{s.name}</option>))}
                </select>
            </div>

import { useState, useEffect } from 'react'

    const [area, setArea] = useState('')
    const [error, setError] = useState('')
    const [assignModal, setAssignModal] = useState(null)
    const [operatorEmail, setOperatorEmail] = useState('')
    const [importFile, setImportFile] = useState(null)

    const role = localStorage.getItem('role')

    async function load() {
        try {
            const data = await api.getSectors()
            setSectors(data || [])
        } catch (err) {
            setError(err.message)
        }
    }

    useEffect(() => { load() }, [])

    // live updates from websocket
    useEffect(() => {
        if (!wsEvents.length) return
        const last = wsEvents[0]
        if (last.event === 'sector:update' && last.data) {
            setSectors(prev => prev.map(s => s.id === last.data.id ? last.data : s))
        }
        if (last.event === 'sector:watered' && last.data?.sector) {
            setSectors(prev => prev.map(s => s.id === last.data.sector.id ? last.data.sector : s))
        }
    }, [wsEvents])

    async function handleCreate(e) {
        e.preventDefault()
        setError('')
        try {
            await api.createSector({ name, area_sqm: parseFloat(area) || 0 })
            setName('')
            setArea('')
            load()
        } catch (err) {
            setError(err.message)
        }
    }

    async function handleDelete(id) {
        if (!confirm('Удалить сектор?')) return
        await api.deleteSector(id)
        load()
    }

    async function handleAssign(sectorId) {
        if (!operatorEmail.trim()) return
        setError('')
        try {
            await api.assignOperator(sectorId, operatorEmail.trim())
            setAssignModal(null)
            setOperatorEmail('')
            load()
        } catch (err) {
            setError(err.message)
        }
    }

    async function handleUnassign(sectorId) {
        try {
            await api.unassignOperator(sectorId)
            load()
        } catch (err) {
            setError(err.message)
        }
    }

    async function handleExport() {
        try {
            const blob = await api.exportSectors()
            const url = URL.createObjectURL(blob)
            const a = document.createElement('a')
            a.href = url
            a.download = 'sectors.csv'
            a.click()
            URL.revokeObjectURL(url)
        } catch (err) {
            setError(err.message)
        }
    }

    async function handleImport() {
        if (!importFile) return
        try {
            const result = await api.importSectors(importFile)
            setImportFile(null)
            alert(`Импортировано секторов: ${result.imported}`)
            load()
        } catch (err) {
            setError(err.message)
        }
    }

    function statusLabel(s) {
        if (s.status === 'drought') return '🏜️ Засуха'
        if (s.status === 'overwatered') return '💧 Перелив'
        return '✓ Норма'
    }

    return (
        <div>
            <h1>📍 Секторы плантации</h1>
            {error && <p className="error">{error}</p>}

                        <input style={{ flex: 1 }} placeholder="Название сектора" value={name} onChange={e => setName(e.target.value)} />
                        <input style={{ flex: 1 }} placeholder="Площадь м²" type="number" value={area} onChange={e => setArea(e.target.value)} />
                        <button className="primary" type="submit">Добавить</button>

                                    <button className="danger" onClick={() => handleDelete(s.id)}>✕</button>

            console.error(err)

            case 'flood_warning': return '🔵'
            case 'health_critical': return '💀'

                            animationDelay: `${i * 0.03}s`,
                            opacity: n.is_read ? 0.6 : 1,

const BASE = '/api'
const WS_BASE = `ws://${window.location.host}/ws`

function getToken() {
    return localStorage.getItem('token')
}

async function request(path, options = {}) {
    const token = getToken()
    const headers = { ...options.headers }

    if (token) {
        headers['Authorization'] = `Bearer ${token}`
    }

    if (!(options.body instanceof FormData)) {
        headers['Content-Type'] = 'application/json'
    }

    const res = await fetch(`${BASE}${path}`, { ...options, headers })
    if (res.status === 204) return null
    const data = await res.json()
    if (!res.ok) throw new Error(data.error || 'request failed')
    return data
}

// websocket connection
export function connectWS(onMessage) {
    let ws = null
    let reconnectTimer = null

    function connect() {
        ws = new WebSocket(WS_BASE)

        ws.onopen = () => console.log('ws connected')

        ws.onmessage = (e) => {
            try {
                const msg = JSON.parse(e.data)
                onMessage(msg)
            } catch (err) {
                console.error('ws parse error', err)
            }
        }

        ws.onclose = () => {
            console.log('ws disconnected, reconnecting in 3s...')
            reconnectTimer = setTimeout(connect, 3000)
        }

        ws.onerror = (err) => {
            console.error('ws error', err)
            ws.close()
        }
    }

    connect()

    return () => {
        clearTimeout(reconnectTimer)
        if (ws) ws.close()
    }
}

import { useState, useEffect, useCallback } from 'react'
import Login from './pages/Login'
import Sectors from './pages/Sectors'
import Plants from './pages/Plants'
import Watering from './pages/Watering'
import Notifications from './pages/Notifications'
import Reports from './pages/Reports'
import { connectWS } from './api'
import './App.css'

export default function App() {
  const [loggedIn, setLoggedIn] = useState(!!localStorage.getItem('token'))
  const [page, setPage] = useState('sectors')
  const [wsEvents, setWsEvents] = useState([])
  const [unreadCount, setUnreadCount] = useState(0)

  const role = localStorage.getItem('role')

  const handleWsMessage = useCallback((msg) => {
    setWsEvents(prev => [msg, ...prev].slice(0, 50))
    if (msg.event === 'notification') {
      setUnreadCount(prev => prev + 1)
    }
  }, [])

  useEffect(() => {
    if (!loggedIn) return
    const disconnect = connectWS(handleWsMessage)
    return disconnect
  }, [loggedIn, handleWsMessage])

  function handleLogout() {
    localStorage.clear()
    setLoggedIn(false)
  }

  if (!loggedIn) {
    return <Login onLogin={() => setLoggedIn(true)} />
  }

  return (
      <div className="app">
        <nav className="navbar">
          <button className={page === 'sectors' ? 'active' : ''} onClick={() => setPage('sectors')}>
            📍 Секторы
          </button>
          <button className={page === 'plants' ? 'active' : ''} onClick={() => setPage('plants')}>
            🌱 Растения
          </button>
          <button className={page === 'watering' ? 'active' : ''} onClick={() => setPage('watering')}>
            💧 Полив
          </button>
          <button className={page === 'notifications' ? 'active' : ''} onClick={() => { setPage('notifications'); setUnreadCount(0) }}>
            🔔 Уведомления {unreadCount > 0 && <span className="notif-badge">{unreadCount}</span>}
          </button>
          {role === 'agronomist' && (
              <button className={page === 'reports' ? 'active' : ''} onClick={() => setPage('reports')}>
                📊 Отчёты
              </button>
          )}

          <div className="spacer" />

          <div className="user-badge">
            👤 {localStorage.getItem('name')} <span>({role === 'agronomist' ? 'агроном' : 'оператор'})</span>
          </div>
          <button className="logout-btn" onClick={handleLogout}>Выйти</button>
        </nav>

        <main className="content">
          {page === 'sectors' && <Sectors wsEvents={wsEvents} />}
          {page === 'plants' && <Plants />}
          {page === 'watering' && <Watering wsEvents={wsEvents} />}
          {page === 'notifications' && <Notifications />}
          {page === 'reports' && <Reports />}
        </main>
      </div>
  )
}

# React + Vite

This template provides a minimal setup to get React working in Vite with HMR and some ESLint rules.

Currently, two official plugins are available:

- [@vitejs/plugin-react](https://github.com/vitejs/vite-plugin-react/blob/main/packages/plugin-react) uses [Oxc](https://oxc.rs)
- [@vitejs/plugin-react-swc](https://github.com/vitejs/vite-plugin-react/blob/main/packages/plugin-react-swc) uses [SWC](https://swc.rs/)

## React Compiler

The React Compiler is not enabled on this template because of its impact on dev & build performances. To add it, see [this documentation](https://react.dev/learn/react-compiler/installation).

## Expanding the ESLint configuration

If you are developing a production application, we recommend using TypeScript with type-aware lint rules enabled. Check out the [TS template](https://github.com/vitejs/vite/tree/main/packages/create-vite/template-react-ts) for information on how to integrate TypeScript and [`typescript-eslint`](https://typescript-eslint.io) in your project.
